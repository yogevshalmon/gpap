        }
        \node[gpapgray,anchor=north west,text width=4.9cm] at (0,3.0)
          {a \textbf{few} clever cores --- each fast on its own, each free to do its own thing};

        % --- GPU
        \node[font=\small,anchor=north west] at (0,1.95) {\textbf{GPU}};
        \foreach \j in {0,...,5} {
          \foreach \i in {0,...,23} {
            \draw[fill=gpapgreen!45,draw=none] (\i*0.2,0.55+\j*0.16) rectangle (\i*0.2+0.15,0.67+\j*0.16);
          }
        }
        \node[gpapgray,anchor=north west,text width=4.9cm] at (0,0.4)
          {\textbf{thousands} of simple cores --- and they move in \keyw{lockstep groups of 32}};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{The deal a GPU offers}
        \small
        Enormous throughput --- \good{\textbf{but only if you hand it thousands of independent
        jobs that all do the same thing.}}
      \end{block}
      \vspace{0.3em}
      \small
      \textbf{And there is a catch.} Fetching from memory is slow. A GPU copes by always having
      \emph{another} job ready to run while one waits.
      \\[0.5em]
      \bad{\textbf{Run out of jobs and the machine simply idles}} --- however fast its arithmetic
      is.
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \centering
  \rem{So the only question that matters for us is: \keyw{how many independent jobs can we
  supply, and how small can we make each one?}}
\end{frame}

      \\[0.2em]
      \small
      Exactly \NC's two numbers per clause --- \emph{how many literals are still alive}, and

  \end{columns}
  \centering

\begin{frame}{One word per clause --- and the update is its own detector}
  \begin{center}
  \begin{tikzpicture}[font=\small]
    % --- packed word
    \draw[fill=gpapblue!12]  (0,1.9)  rectangle (7.0,2.6);
    \draw[fill=gpapgreen!14] (7.0,1.9) rectangle (10.4,2.6);
    \node at (3.5,2.25) {$\sum$ surviving literal indices $(\bmod\ 2^{F})$};
    \node at (8.7,2.25) {\texttt{sat\_count}};
    \node[font=\scriptsize,gpapgray,anchor=south west] at (0,2.65) {bit 31};
    \node[font=\scriptsize,gpapgray,anchor=south east] at (10.4,2.65) {bit 0};

    % --- the update
    \draw[ar] (2.6,1.85) -- (2.6,0.5);
    \node[bxn,text width=5.6cm,font=\small,minimum height=1.0cm] (upd) at (2.6,-0.10)
      {falsifying $\lnot\ell$ is \keyw{one subtraction},\\ hence \keyw{one \texttt{atomic\_add}}:\\[0.2em]
       \texttt{st} $\mathrel{-}=\ 1 + (\lnot\ell \ll \mathrm{SHIFT})$};

    % --- the fan
    \node[bxr,text width=4.2cm,minimum height=0.7cm] (o1) at (9.0,0.95)  {old count $=1$: \bad{conflict}};
    \node[bxg,text width=4.2cm,minimum height=0.95cm] (o2) at (9.0,-0.10)
      {old count $=2$: \good{unit},\\ forced literal $=$ old sum $-\ \lnot\ell$};
    \node[bxn,text width=4.2cm,minimum height=0.7cm] (o3) at (9.0,-1.15) {otherwise: nothing to do};
    \draw[ar] (upd.east) -- (o1.west);
    \draw[ar] (upd.east) -- (o2.west);
    \draw[ar] (upd.east) -- (o3.west);
    \node[font=\scriptsize,gpapblue,anchor=south,fill=white,inner sep=1pt] at (6.4,1.55)
      {\textbf{the returned OLD word is the whole detection logic}};
  \end{tikzpicture}
  \end{center}
  \vspace{0.4em}
  \centering
  \rem{The \keyw{assignment} is the same trick: a bitmap where $\ell$ and $\lnot\ell$ share a word,
  so one \texttt{atomic\_or} returns duplicate \emph{and} contradiction together.
  \\ On a CPU, XOR and sum are equivalent --- \NC{} used XOR. On a GPU they are not: the sum makes
  the update an atomic whose \emph{return value} resolves every race --- no locks, no
  \texttt{shfl}/\texttt{ballot}.}
\end{frame}

\begin{frame}{The architecture: one assignment per warp}
  \begin{columns}[T]
    % ============ LEFT: small formula, K huge ============
    \begin{column}{0.48\textwidth}
      \centering
      \textbf{\color{gpapblue}Small formula} $\Rightarrow$ \textbf{${\sim}100$k assignments fit}
      \\[0.4em]
      \begin{tikzpicture}[font=\tiny]
        \foreach \y in {0,1,2,3,4} { \draw[fill=gpapblue!25] (0,\y*0.26) rectangle (4.6,\y*0.26+0.19); }
        \node[gpapgray,anchor=north] at (2.3,-0.05) {$\vdots$ \ tens of thousands more};
      \end{tikzpicture}
      \\[0.5em]
      \small
      \good{\textbf{The GPU is full}} --- while one assignment waits on memory, thousands of
      others run. \bad{Extra effort per assignment buys nothing.}
      \\[0.5em]
      \begin{tikzpicture}
        \foreach \i in {0,...,15} { \draw[fill=black!10] (\i*0.2,0) rectangle (\i*0.2+0.16,0.2); }
        \draw[fill=gpapblue!85] (0,0) rectangle (0.16,0.2);
      \end{tikzpicture}
      \\[0.2em]
      \textbf{SCALAR}: \keyw{one lane of the 32} does the row's work --- the other 31 rest
    \end{column}
    % ============ RIGHT: big formula, K tiny ============
    \begin{column}{0.48\textwidth}
      \centering
      \textbf{\color{gpapgreen}Big formula} $\Rightarrow$ \textbf{only hundreds fit}
      \\[0.4em]
      \begin{tikzpicture}[font=\tiny]
        \foreach \y in {3,4} { \draw[fill=gpapgreen!35] (0,\y*0.26) rectangle (4.6,\y*0.26+0.19); }
        \draw[dashed,gpapgray] (0,0) rectangle (4.6,0.72);
        \node[gpapgray] at (2.3,0.36) {most of the GPU: nothing to run};
      \end{tikzpicture}
      \\[0.5em]
      \small
      \bad{\textbf{The GPU is starving}} --- too few assignments to keep it busy.
      \good{\textbf{Now we \emph{want} more work per assignment.}}
      \\[0.5em]
      \begin{tikzpicture}
        \foreach \i in {0,...,15} { \draw[fill=gpapgreen!70] (\i*0.2,0) rectangle (\i*0.2+0.16,0.2); }
      \end{tikzpicture}
      \\[0.2em]
      \textbf{WAVE}: \keyw{all 32 lanes} pitch in on the \emph{same} assignment
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \centering
  \keyw{One rule, measured}: WAVE when there are too few assignments to fill the GPU, SCALAR once
  there are plenty. \rem{Same data, identical results either way.}
\end{frame}
